SELECT 
    sensor_id,
    subsystem_name,
    COUNT(CASE WHEN error_code IS NOT NULL THEN 1 END) AS error_count,
    ROUND(AVG(pressure_psi)::numeric, 2) AS avg_pressure,
    ROUND(STDDEV(temperature_c)::numeric, 2) AS temp_variance
FROM braking_telemetry_logs
WHERE timestamp >= NOW() - INTERVAL '7 days'
GROUP BY sensor_id, subsystem_name
HAVING COUNT(CASE WHEN error_code IS NOT NULL THEN 1 END) > 5
ORDER BY error_count DESC;

eda_diagnostics.py Python 3.11 / Pandas 
import pandas as pd
import numpy as np

# Load braking sensor log telemetry dataset
df = pd.DataFrame({
    'timestamp': pd.date_range(start='2026-01-01', periods=5, freq='H'),
    'sensor_id': ['BRK-9021', 'BRK-9021', 'BRK-4011', 'BRK-3304', 'BRK-9021'],
    'pressure_psi': [142.5, 185.0, 128.1, 155.0, 192.3],
    'is_fault': [False, True, False, False, True]
})

# Calculate anomaly flags based on threshold
pressure_mean = df['pressure_psi'].mean()
pressure_std = df['pressure_psi'].std()
df['anomaly_flag'] = df['pressure_psi'] > (pressure_mean + 1.2 * pressure_std)

print("--- Diagnostic Summary ---")
print(f"Total Telemetry Readings: {len(df)}")
print(f"Anomalies Detected: {df['anomaly_flag'].sum()}")
print(df[['sensor_id', 'pressure_psi', 'anomaly_flag']])
